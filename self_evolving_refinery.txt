    layer_1_ingestion:

      purpose: "Feed everything into the learning system"
      sources:

        - "Entity recognition (drugs, proteins, symptoms)"
        - "Relationship mapping (cause → effect)"
        - "Vector embedding (for semantic search)"
      
      output: "Unified knowledge graph stored in Neo4j + Qdrant"
    
    layer_2_llm_forge:
      name: "Custom LLM Training System"
      purpose: "Build disease-specific language models"
      
      base_models:
        - "Qwen2.5:72b (general medical)"
        - "BioGPT (specialized biomedical)"
        - "PubMedBERT (paper understanding)"
      
      fine_tuning:
        method: "LoRA (Low-Rank Adaptation)"
        dataset: "All ingested knowledge (Layer 1)"
        compute: "Local GPUs + TPU pods (if needed)"
        cost: "$0 local, $500/month cloud"
        
      custom_models_created:
        - name: "NeuroPath-72b"
          specialty: "Neurological disease pathways"
          training_data: "10,000 neuro papers + her case notes"
          
        - name: "PainPredict-32b"
          specialty: "Chronic pain mechanisms"
          training_data: "5,000 pain studies + symptom logs"
          
        - name: "DrugRepurpose-70b"
          specialty: "Finding off-label drug uses"
          training_data: "DrugBank + 50K clinical trials"
          
        - name: "QuantumMed-405b"
          specialty: "Quantum computing + drug design"
          training_data: "Quantum chemistry papers + sim results"
      
      evolution_mechanism:
        - "Every 1000 research papers ingested → retrain"
        - "Agent feedback loop: 'This prediction was wrong' → improve"
        - "A/B testing: Old model vs new model on same query"
        - "Keep best performing model, archive old versions"
    
    layer_3_self_correction:
      name: "Error Detection & Auto-Fix System"
      purpose: "Agents check each other's work and self-improve"
      
      validation_pipeline:
        step_1_peer_review:
          - "Agent A makes claim: 'Memantine may help'"
          - "Agent B verifies: Checks 10 papers, 3 trials"
          - "Agent C cross-checks: Runs quantum sim"
          - "Consensus: If 2/3 agree → accept, else → reject"
        
        step_2_fact_checking:
          - "Every claim linked to source paper"
          - "Automated DOI lookup + full-text verification"
          - "If paper retracted → flag claim as invalid"
          - "Update knowledge graph automatically"
        
        step_3_outcome_tracking:
          - "If recommendation tried → track result"
          - "If it helped → reinforce that pathway"
          - "If it failed → deprecate that hypothesis"
          - "Learn from failures faster than successes"
      
      auto_correction:
        triggers:
          - "Agent makes prediction → wrong → retrain on error"
          - "Paper contradicts knowledge graph → update graph"
          - "New trial data invalidates old hypothesis → revise"
        
        learning_rate: "Aggressive (update within 1 hour of error)"
    
    layer_4_evolution_engine:
      name: "Genetic Algorithm for Agent Improvement"
      purpose: "Agents compete, best ones reproduce"
      
      fitness_function:
        metrics:
          - "Accuracy (% of correct predictions)"
          - "Speed (time to find answer)"
          - "Novelty (how unique is the insight)"
          - "Actionability (can we use this today?)"
          - "Cost (is it affordable?)"
        
        scoring: |
          fitness = (0.4 × accuracy) + (0.2 × speed) + 
                    (0.2 × novelty) + (0.15 × actionability) + 
                    (0.05 × cost_efficiency)
      
      selection:
        - "Top 10% agents by fitness score"
        - "Clone their parameters (LoRA weights)"
        - "Mutate slightly (add random noise)"
        - "Replace bottom 10% agents"
      
      crossover:
        - "Take best 2 agents in same department"
        - "Combine their approaches (ensemble)"
        - "Create hybrid agent with both strengths"
      
      mutation:
        - "Random parameter tweaks (5% chance)"
        - "Try new search strategies"
        - "Test unconventional hypotheses"
      
      generations:
        frequency: "Every 7 days (one research cycle)"
        goal: "Each generation 5-10% better than last"
    
    layer_5_obsidian_integration:
      name: "Bidirectional Learning from Vault"
      purpose: "Obsidian becomes the agent's long-term memory"
      
      read_operations:
        - "Agents query Obsidian for past research"
        - "Dataview queries: 'Show me all papers on IL-6'"
        - "Graph traversal: 'What's connected to autoimmune?'"
        - "Embeddings: 'Find similar case reports'"
      
      write_operations:
        - "Agents add discoveries to vault"
        - "Auto-generate markdown files"
        - "Link to related notes (bidirectional)"
        - "Update STATE.yaml with progress"
      
      learning_loop:
        - "Agent reads old note: 'We tried X, it failed'"
        - "Agent learns: 'Don't recommend X again'"
        - "Agent finds new paper: 'X works if combined with Y'"
        - "Agent updates note: 'X + Y hypothesis'"
        - "Vault evolves with agent's knowledge"
      
      obsidian_as_brain:
        analogy: |
          Obsidian = Long-term memory (facts, papers, history)
          Agents = Working memory (active research)
          Knowledge graph = Neural connections
          Auto-linking = Synapse formation
          
          The vault literally becomes the swarm's brain.
